# /etc/systemd/system/gnss.service
[Unit]
Description=GNSS Java Base Station Auto Start
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=debian
WorkingDirectory=/opt/gnss
ExecStart=/opt/gnss/start_gnss.sh
Restart=always
RestartSec=10
SuccessExitStatus=143

[Install]
WantedBy=multi-user.target
# /etc/systemd/system/serial-bridge.service
[Unit]
Description=Serial-to-LAN Bridge
After=network.target

[Service]
ExecStart=/bin/bash -c "sleep 10; /usr/local/bin/serial_bridge"
Restart=always
User=root

[Install]
WantedBy=multi-user.target
